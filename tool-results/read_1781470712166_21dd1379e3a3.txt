     8→// ─── In-memory Rate Limiter for Seed Endpoint ───────────────────
     9→const seedRateLimitMap = new Map<string, { count: number; resetAt: number }>();
    10→const SEED_RATE_LIMIT_MAX = 3;       // max attempts per window
    11→const SEED_RATE_LIMIT_WINDOW = 60000; // 1 minute window
    12→
    13→function checkSeedRateLimit(ip: string): { allowed: boolean; retryAfterMs: number } {
    14→  const now = Date.now();
    15→  const entry = seedRateLimitMap.get(ip);
    16→
    17→  if (!entry || now > entry.resetAt) {
    18→    // No entry or window expired — start fresh
    19→    seedRateLimitMap.set(ip, { count: 1, resetAt: now + SEED_RATE_LIMIT_WINDOW });
    20→    return { allowed: true, retryAfterMs: 0 };
    21→  }
    22→
    23→  if (entry.count >= SEED_RATE_LIMIT_MAX) {
    24→    return { allowed: false, retryAfterMs: entry.resetAt - now };
    25→  }
    26→
    27→  entry.count += 1;
    28→  return { allowed: true, retryAfterMs: 0 };
    29→}

    49→    where: { email: data.email, restaurantId: data.restaurantId },
    50→  });
    51→  if (existing) {
    52→    // Update password in case it changed between seeds

    57→    return existing;
    58→  }
    59→  return db.customer.create({ data });
    60→}

   576→export async function DELETE(request: Request) {
   577→  try {
   578→    const admin = await authenticateAdmin(request);
   579→    if (!admin) {
   580→      return NextResponse.json({ error: "Non autorisé" }, { status: 401 });
   581→    }
   582→    if (!hasRole(admin.role, ["admin"])) {
   583→      return NextResponse.json({ error: "Accès refusé" }, { status: 403 });
   584→    }
   585→
   586→    await db.$transaction([
   587→      db.expense.deleteMany(),
   588→      db.quote.deleteMany(),
   589→      db.invoice.deleteMany(),
   590→      db.review.deleteMany(),
   591→      db.order.deleteMany(),
   592→      db.reservation.deleteMany(),
   593→      db.staff.deleteMany(),
   594→      db.menuItem.deleteMany(),
   595→      db.driver.deleteMany(),
   596→      db.customer.deleteMany(),
   597→      db.admin.deleteMany(),
   598→      db.restaurantConfig.deleteMany(),
   599→      db.restaurant.deleteMany(),
   600→      db.platformAdmin.deleteMany(),
   601→    ]);
   602→
   603→    return NextResponse.json({ success: true, message: "Base de données vidée avec succès" });
   604→  } catch (error) {
