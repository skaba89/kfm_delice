    50→/* ─── custom tooltip component ─── */
    51→interface CustomTooltipProps {
    52→  active?: boolean;
    53→  payload?: Array<{ value: number; name: string; dataKey: string; color?: string }>;
    54→  label?: string;
    55→  isCurrency?: boolean;
    56→}
    57→
    58→function ChartTooltip({ active, payload, label, isCurrency = false }: CustomTooltipProps) {
    59→  if (!active || !payload?.length) return null;
    60→  return (
    61→    <div className="bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-lg px-3 py-2 shadow-lg text-sm">
    62→      {label && <p className="text-gray-500 dark:text-gray-400 mb-1 font-medium">{label}</p>}
    63→      {payload.map((entry, i) => (
    64→        <p key={i} className="flex items-center gap-2">
    65→          <span className="w-2 h-2 rounded-full inline-block" style={{ background: entry.color }} />
    66→          <span className="text-gray-600 dark:text-gray-300">{entry.name}: </span>
    67→          <span className="font-bold text-gray-900 dark:text-gray-100">
    68→            {isCurrency || entry.dataKey === "revenue" ? formatPrice(entry.value) : entry.value}
    69→          </span>
    70→        </p>
    71→      ))}
    72→    </div>
    73→  );
    74→}
