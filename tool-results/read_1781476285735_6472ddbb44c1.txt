     8→     8→// ─── In-memory Rate Limiter for Seed Endpoint ───────────────────
     9→     9→const seedRateLimitMap = new Map<string, { count: number; resetAt: number }>();
    10→    10→const SEED_RATE_LIMIT_MAX = 3;       // max attempts per window
    11→    11→const SEED_RATE_LIMIT_WINDOW = 60000; // 1 minute window
    12→    12→
    13→    13→function checkSeedRateLimit(ip: string): { allowed: boolean; retryAfterMs: number } {
    14→    14→  const now = Date.now();
    15→    15→  const entry = seedRateLimitMap.get(ip);
    16→    16→
    17→    17→  if (!entry || now > entry.resetAt) {
    18→    18→    // No entry or window expired — start fresh
    19→    19→    seedRateLimitMap.set(ip, { count: 1, resetAt: now + SEED_RATE_LIMIT_WINDOW });
    20→    20→    return { allowed: true, retryAfterMs: 0 };
    21→    21→  }
    22→    22→
    23→    23→  if (entry.count >= SEED_RATE_LIMIT_MAX) {
    24→    24→    return { allowed: false, retryAfterMs: entry.resetAt - now };
    25→    25→  }
    26→    26→
    27→    27→  entry.count += 1;
    28→    28→  return { allowed: true, retryAfterMs: 0 };
    29→    29→}
