     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useMemo } from "react";
     4→import { useRouter } from "next/navigation";
     5→import { motion, AnimatePresence } from "framer-motion";
     6→import {
     7→  Plus, Minus, Trash2, ShoppingCart, ChevronRight, RefreshCw,
     8→  ShoppingBag, Tag, UtensilsCrossed, Package, Truck, Star,
     9→  ArrowLeft, CheckCircle2, MapPin, CreditCard, StickyNote,
    10→  Search,
    11→} from "lucide-react";
    12→import { Button } from "@/components/ui/button";
    13→import { Card, CardContent } from "@/components/ui/card";
    14→import { Input } from "@/components/ui/input";
    15→import { Badge } from "@/components/ui/badge";
    16→import { Textarea } from "@/components/ui/textarea";
    17→import { Separator } from "@/components/ui/separator";
    18→import {
    19→  Sheet, SheetContent, SheetHeader, SheetTitle, SheetFooter, SheetDescription,
    20→} from "@/components/ui/sheet";
    21→import { PublicNavbar } from "@/components/PublicNavbar";
    22→import { PublicFooter } from "@/components/PublicFooter";
    23→import { useAuth } from "@/lib/auth-context";
    24→import { notify } from "@/lib/notifications";
    25→import { getTier } from "@/lib/hooks/use-loyalty";
    26→import { MENU_CATS, formatPrice, paymentLabels, orderTypeLabels, isRestaurantOpen, RESTO } from "@/lib/constants";
    27→import type { MenuItemDB } from "@/lib/types";
    28→
    29→// ─── Cart types ───────────────────────────────────────────────
    30→interface CartItem {
    31→  item: MenuItemDB;
    32→  qty: number;
    33→}
    34→
    35→type OrderType = "dine_in" | "takeaway" | "delivery";
    36→type AppStep = "menu" | "checkout" | "success";
    37→
    38→// ─── Page Component ───────────────────────────────────────────
    39→export default function MenuOrderingPage() {
    40→  const { customer, driver } = useAuth();
    41→  const router = useRouter();
    42→
    43→  // Menu data
    44→  const [menuItems, setMenuItems] = useState<MenuItemDB[]>([]);
    45→  const [loading, setLoading] = useState(true);
    46→  const [activeCat, setActiveCat] = useState("entrees");
    47→  const [searchQuery, setSearchQuery] = useState("");
    48→
    49→  // Cart
    50→  const [cart, setCart] = useState<CartItem[]>([]);
    51→  const [cartOpen, setCartOpen] = useState(false);
    52→
    53→  // Order flow
    54→  const [orderType, setOrderType] = useState<OrderType>("delivery");
    55→  const [step, setStep] = useState<AppStep>("menu");
    56→  const [submitting, setSubmitting] = useState(false);
    57→  const [orderId, setOrderId] = useState<string | null>(null);
    58→
    59→  // Checkout form
    60→  const [guestName, setGuestName] = useState("");
    61→  const [guestPhone, setGuestPhone] = useState("");
    62→  const [guestEmail, setGuestEmail] = useState("");
    63→  const [address, setAddress] = useState("");
    64→  const [paymentMethod, setPaymentMethod] = useState<string>("cash");
    65→  const [note, setNote] = useState("");
    66→
    67→  const deliveryFee = 5000;
    68→
    69→  // ─── Fetch menu ──────────────────────────────────────────────
    70→  useEffect(() => {
    71→    fetch("/api/menu?limit=1000")
    72→      .then(r => r.json())
    73→      .then(d => {
    74→        setMenuItems(Array.isArray(d) ? d : (d.data || []));
    75→        setLoading(false);
    76→      })
    77→      .catch(() => setLoading(false));
    78→  }, []);
    79→
    80→  // Pre-fill if logged in
    81→  useEffect(() => {
    82→    if (customer) {
    83→      setGuestName(customer.name || "");
    84→      setGuestPhone(customer.phone || "");
    85→      setAddress(customer.address || "");
    86→    }
    87→  }, [customer]);
    88→
    89→  // ─── Cart helpers ────────────────────────────────────────────
    90→  const addToCart = useCallback((item: MenuItemDB) => {
    91→    setCart(prev => {
    92→      const existing = prev.find(c => c.item.id === item.id);
    93→      if (existing) return prev.map(c => c.item.id === item.id ? { ...c, qty: c.qty + 1 } : c);
    94→      return [...prev, { item, qty: 1 }];
    95→    });
    96→  }, []);
    97→
    98→  const removeFromCart = useCallback((itemId: string) => {
    99→    setCart(prev => prev.filter(c => c.item.id !== itemId));
   100→  }, []);
   101→
   102→  const updateQty = useCallback((itemId: string, qty: number) => {
   103→    if (qty <= 0) {
   104→      setCart(prev => prev.filter(c => c.item.id !== itemId));
   105→      return;
   106→    }
   107→    setCart(prev => prev.map(c => c.item.id === itemId ? { ...c, qty } : c));
   108→  }, []);
   109→
   110→  // ─── Totals ──────────────────────────────────────────────────
   111→  const cartSubtotal = useMemo(() => cart.reduce((sum, c) => sum + c.item.price * c.qty, 0), [cart]);
   112→  const cartCount = useMemo(() => cart.reduce((sum, c) => sum + c.qty, 0), [cart]);
   113→
   114→  const discountPercent = customer ? parseFloat(getTier(customer.loyaltyPoints).discount) || 0 : 0;
   115→  const discountAmount = Math.round(cartSubtotal * (discountPercent / 100));
   116→  const cartTotal = cartSubtotal - discountAmount;
   117→  const finalTotal = cartTotal + (orderType === "delivery" ? deliveryFee : 0);
   118→
   119→  // ─── Filtered items ──────────────────────────────────────────
   120→  const filteredItems = useMemo(() => {
   121→    return menuItems
   122→      .filter(i => i.category === activeCat && i.available)
   123→      .filter(i => {
   124→        if (!searchQuery) return true;
   125→        const q = searchQuery.toLowerCase();
   126→        return i.name.toLowerCase().includes(q) || i.description?.toLowerCase().includes(q);
   127→      });
   128→  }, [menuItems, activeCat, searchQuery]);
   129→
   130→  // ─── Submit order ────────────────────────────────────────────
   131→  const submitOrder = useCallback(async () => {
   132→    if (cart.length === 0) return;
   133→    if (!guestName.trim()) { notify.error("Veuillez entrer votre nom"); return; }
   134→    if (!guestPhone.trim()) { notify.error("Veuillez entrer votre téléphone"); return; }
   135→    if (orderType === "delivery" && !address.trim()) { notify.error("Veuillez entrer votre adresse de livraison"); return; }
   136→
   137→    setSubmitting(true);
   138→    try {
   139→      const items = cart.map(c => ({ name: c.item.name, price: c.item.price, qty: c.qty }));
   140→      const res = await fetch("/api/orders", {
   141→        method: "POST",
   142→        headers: { "Content-Type": "application/json" },
   143→        body: JSON.stringify({
   144→          customerName: guestName,
   145→          phone: guestPhone,
   146→          items: JSON.stringify(items),
   147→          total: finalTotal,
   148→          orderType,
   149→          paymentMethod,
   150→          deliveryAddress: orderType === "delivery" ? address : "",
   151→          deliveryFee: orderType === "delivery" ? deliveryFee : 0,
   152→          discount: discountAmount,
   153→          note,
   154→        }),
   155→      });
   156→      if (res.ok) {
   157→        const data = await res.json();
   158→        setOrderId(data.id || null);
   159→        setCart([]);
   160→        setStep("success");
   161→        setCartOpen(false);
   162→        notify.success("Commande passée avec succès !");
   163→      } else {
   164→        const err = await res.json();
   165→        notify.error(err.error || "Erreur lors de la commande");
   166→      }
   167→    } catch {
   168→      notify.error("Erreur de connexion");
   169→    } finally {
   170→      setSubmitting(false);
   171→    }
   172→  }, [cart, guestName, guestPhone, address, orderType, paymentMethod, finalTotal, deliveryFee, discountAmount, note]);
   173→
   174→  const restaurantOpen = isRestaurantOpen();
   175→
   176→  // ─── Render ──────────────────────────────────────────────────
   177→  return (
   178→    <div className="min-h-screen flex flex-col bg-gray-50 dark:bg-gray-950">
   179→      <PublicNavbar
   180→        onAdminClick={() => router.push("/admin/login")}
   181→        onCustomerClick={() => { if (customer) router.push("/client"); else router.push("/client/login"); }}
   182→        onDriverClick={() => { if (driver) router.push("/driver"); else router.push("/driver/login"); }}
   183→        customer={customer}
   184→      />
   185→
   186→      <main className="flex-1 pt-16">
   187→        <AnimatePresence mode="wait">
   188→          {step === "menu" && (
   189→            <motion.div key="menu" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}>
   190→              {/* ── Hero Banner ────────────────────────────── */}
   191→              <div className="relative bg-gradient-to-br from-orange-600 via-red-600 to-orange-700 overflow-hidden">
   192→                <div className="absolute inset-0 bg-black/20" />
   193→                <div className="absolute -top-20 -right-20 w-72 h-72 bg-orange-400/20 rounded-full blur-3xl" />
   194→                <div className="absolute -bottom-16 -left-16 w-56 h-56 bg-red-400/20 rounded-full blur-3xl" />
   195→                <div className="relative max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-10 sm:py-14">
   196→                  <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.5 }}>
   197→                    <Badge className="bg-white/20 text-white border-0 mb-3 backdrop-blur-sm">
   198→                      <ShoppingBag className="w-3 h-3 mr-1" /> Commande en ligne
   199→                    </Badge>
   200→                    <h1 className="text-3xl sm:text-4xl lg:text-5xl font-extrabold text-white mb-2">
   201→                      Commandez en ligne
   202→                    </h1>
   203→                    <p className="text-white/80 text-base sm:text-lg mb-6">
   204→                      {RESTO.name} &bull; Livraison à Conakry
   205→                    </p>
   206→
   207→                    {/* Order type toggle */}
   208→                    <div className="flex flex-wrap gap-2">
   209→                      {([
   210→                        { type: "dine_in" as OrderType, icon: UtensilsCrossed, label: "Sur place" },
   211→                        { type: "takeaway" as OrderType, icon: Package, label: "À emporter" },
   212→                        { type: "delivery" as OrderType, icon: Truck, label: "Livraison" },
   213→                      ]).map(({ type, icon: Icon, label }) => (
   214→                        <button
   215→                          key={type}
   216→                          onClick={() => setOrderType(type)}
   217→                          className={`flex items-center gap-2 px-5 py-2.5 rounded-xl text-sm font-semibold transition-all ${
   218→                            orderType === type
   219→                              ? "bg-white text-orange-700 shadow-lg shadow-black/20"
   220→                              : "bg-white/15 text-white hover:bg-white/25 backdrop-blur-sm"
   221→                          }`}
   222→                        >
   223→                          <Icon className="w-4 h-4" /> {label}
   224→                        </button>
   225→                      ))}
   226→                    </div>
   227→
   228→                    {!restaurantOpen && (
   229→                      <div className="mt-4 px-4 py-3 bg-amber-500/20 border border-amber-400/30 rounded-xl backdrop-blur-sm">
   230→                        <p className="text-amber-100 text-sm font-medium">
   231→                          ⏰ Le restaurant est actuellement fermé. Heures d&apos;ouverture : 11h - 23h.
   232→                          {orderType === "delivery" && " Vous pouvez quand même passer une commande pour plus tard."}
   233→                        </p>
   234→                      </div>
   235→                    )}
   236→                  </motion.div>
   237→                </div>
   238→              </div>
   239→
   240→              {/* ── Loyalty banner ──────────────────────────── */}
   241→              {discountPercent > 0 && (
   242→                <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 mt-4">
   243→                  <div className="flex items-center gap-2 p-3 bg-emerald-50 dark:bg-emerald-900/20 border border-emerald-200 dark:border-emerald-800 rounded-xl">
   244→                    <Tag className="w-4 h-4 text-emerald-600 dark:text-emerald-400 shrink-0" />
   245→                    <p className="text-sm text-emerald-700 dark:text-emerald-300">
   246→                      Réduction fidélité de <strong>{discountPercent}%</strong> appliquée automatiquement !
   247→                    </p>
   248→                  </div>
   249→                </div>
   250→              )}
   251→
   252→              {/* ── Main content: sidebar + grid ────────────── */}
   253→              <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
   254→                {/* Mobile category scroll */}
   255→                <div className="lg:hidden flex gap-2 overflow-x-auto pb-3 scrollbar-hide -mx-4 px-4">
   256→                  {MENU_CATS.map(c => (
   257→                    <button
   258→                      key={c.id}
   259→                      onClick={() => setActiveCat(c.id)}
   260→                      className={`flex items-center gap-2 px-4 py-2.5 rounded-full text-sm font-semibold whitespace-nowrap transition-all shrink-0 ${
   261→                        activeCat === c.id
   262→                          ? "bg-gradient-to-r from-orange-500 to-red-500 text-white shadow-lg shadow-orange-500/25"
   263→                          : "bg-white dark:bg-gray-800 text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-700 border border-gray-200 dark:border-gray-700"
   264→                      }`}
   265→                    >
   266→                      <c.icon className="w-4 h-4" /> {c.name}
   267→                    </button>
   268→                  ))}
   269→                </div>
   270→
   271→                <div className="flex gap-6">
   272→                  {/* Desktop sidebar */}
   273→                  <aside className="hidden lg:block w-56 shrink-0">
   274→                    <div className="sticky top-24 space-y-1">
   275→                      {MENU_CATS.map(c => (
   276→                        <button
   277→                          key={c.id}
   278→                          onClick={() => setActiveCat(c.id)}
   279→                          className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-medium transition-all text-left ${
   280→                            activeCat === c.id
   281→                              ? "bg-gradient-to-r from-orange-500 to-red-500 text-white shadow-lg shadow-orange-500/25"
   282→                              : "text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-800"
   283→                          }`}
   284→                        >
   285→                          <c.icon className="w-4 h-4" /> {c.name}
   286→                          {menuItems.filter(i => i.category === c.id && i.available).length > 0 && (
   287→                            <span className={`ml-auto text-xs ${activeCat === c.id ? "text-white/70" : "text-gray-400"}`}>
   288→                              {menuItems.filter(i => i.category === c.id && i.available).length}
   289→                            </span>
   290→                          )}
   291→                        </button>
   292→                      ))}
   293→                    </div>
   294→                  </aside>
   295→
   296→                  {/* Items grid */}
   297→                  <div className="flex-1 min-w-0">
   298→                    {/* Search */}
   299→                    <div className="relative mb-4">
   300→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   301→                      <Input
   302→                        value={searchQuery}
   303→                        onChange={e => setSearchQuery(e.target.value)}
   304→                        placeholder="Rechercher un plat..."
   305→                        className="pl-9 bg-white dark:bg-gray-800 border-gray-200 dark:border-gray-700"
   306→                      />
   307→                    </div>
   308→
   309→                    {loading ? (
   310→                      <div className="flex justify-center py-20">
   311→                        <RefreshCw className="w-8 h-8 text-orange-500 animate-spin" />
   312→                      </div>
   313→                    ) : filteredItems.length === 0 ? (
   314→                      <div className="text-center py-16">
   315→                        <ShoppingBag className="w-12 h-12 text-gray-300 dark:text-gray-600 mx-auto mb-3" />
   316→                        <p className="text-gray-500 dark:text-gray-400">Aucun article trouvé</p>
   317→                      </div>
   318→                    ) : (
   319→                      <div className="grid sm:grid-cols-2 xl:grid-cols-3 gap-4">
   320→                        {filteredItems.map((item, idx) => {
   321→                          const inCart = cart.find(c => c.item.id === item.id);
   322→                          return (
   323→                            <motion.div
   324→                              key={item.id}
   325→                              initial={{ opacity: 0, y: 20 }}
   326→                              animate={{ opacity: 1, y: 0 }}
   327→                              transition={{ duration: 0.3, delay: Math.min(idx * 0.05, 0.3) }}
   328→                            >
   329→                              <Card className="overflow-hidden hover:shadow-lg transition-all duration-300 group dark:bg-gray-800 dark:border-gray-700">
   330→                                <div className="h-40 overflow-hidden relative">
   331→                                  {item.image ? (
   332→                                    <img
   333→                                      src={item.image}
   334→                                      alt={item.name}
   335→                                      className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
   336→                                    />
   337→                                  ) : (
   338→                                    <div className="w-full h-full bg-gradient-to-br from-orange-100 to-red-100 dark:from-orange-900/30 dark:to-red-900/30 flex items-center justify-center">
   339→                                      <UtensilsCrossed className="w-8 h-8 text-orange-300 dark:text-orange-700" />
   340→                                    </div>
   341→                                  )}
   342→                                  {item.badge && (
   343→                                    <Badge className="absolute top-2 right-2 bg-gradient-to-r from-orange-500 to-red-500 text-white text-[10px]">
   344→                                      {item.badge}
   345→                                    </Badge>
   346→                                  )}
   347→                                  {item.popular && (
   348→                                    <Badge className="absolute top-2 left-2 bg-amber-500 text-white text-[10px]">
   349→                                      <Star className="w-3 h-3 mr-0.5 fill-white" /> Populaire
   350→                                    </Badge>
   351→                                  )}
   352→                                </div>
   353→                                <CardContent className="p-4">
   354→                                  <div className="flex justify-between items-start mb-1">
   355→                                    <h3 className="font-bold text-gray-900 dark:text-gray-100 text-sm leading-tight flex-1 min-w-0 mr-2">{item.name}</h3>
   356→                                    <span className="text-sm font-extrabold bg-gradient-to-r from-orange-500 to-red-500 bg-clip-text text-transparent whitespace-nowrap">
   357→                                      {formatPrice(item.price)}
   358→                                    </span>
   359→                                  </div>
   360→                                  <p className="text-xs text-gray-500 dark:text-gray-400 line-clamp-2 mb-3">{item.description}</p>
   361→                                  {inCart ? (
   362→                                    <div className="flex items-center justify-between">
   363→                                      <span className="text-xs text-orange-600 dark:text-orange-400 font-medium">
   364→                                        {formatPrice(inCart.item.price * inCart.qty)}
   365→                                      </span>
   366→                                      <div className="flex items-center gap-1.5">
   367→                                        <button
   368→                                          onClick={() => updateQty(item.id, inCart.qty - 1)}
   369→                                          className="w-8 h-8 rounded-lg bg-gray-100 dark:bg-gray-700 flex items-center justify-center text-gray-600 dark:text-gray-400 hover:bg-gray-200 dark:hover:bg-gray-600 transition-colors"
   370→                                        >
   371→                                          <Minus className="w-3.5 h-3.5" />
   372→                                        </button>
   373→                                        <span className="text-sm font-bold w-6 text-center dark:text-gray-200">{inCart.qty}</span>
   374→                                        <button
   375→                                          onClick={() => updateQty(item.id, inCart.qty + 1)}
   376→                                          className="w-8 h-8 rounded-lg bg-gradient-to-r from-orange-500 to-red-500 flex items-center justify-center text-white hover:shadow-md transition-all"
   377→                                        >
   378→                                          <Plus className="w-3.5 h-3.5" />
   379→                                        </button>
   380→                                      </div>
   381→                                    </div>
   382→                                  ) : (
   383→                                    <Button
   384→                                      size="sm"
   385→                                      onClick={() => addToCart(item)}
   386→                                      className="w-full bg-gradient-to-r from-orange-500 to-red-500 hover:from-orange-600 hover:to-red-600 text-white text-xs rounded-lg"
   387→                                    >
   388→                                      <Plus className="w-3.5 h-3.5 mr-1" /> Ajouter
   389→                                    </Button>
   390→                                  )}
   391→                                </CardContent>
   392→                              </Card>
   393→                            </motion.div>
   394→                          );
   395→                        })}
   396→                      </div>
   397→                    )}
   398→                  </div>
   399→                </div>
   400→              </div>
   401→
   402→              {/* ── Floating cart bar ───────────────────────── */}
   403→              <AnimatePresence>
   404→                {cartCount > 0 && step === "menu" && (
   405→                  <motion.div
   406→                    initial={{ y: 100, opacity: 0 }}
   407→                    animate={{ y: 0, opacity: 1 }}
   408→                    exit={{ y: 100, opacity: 0 }}
   409→                    transition={{ type: "spring", stiffness: 300, damping: 30 }}
   410→                    className="fixed bottom-0 left-0 right-0 z-40 p-4"
   411→                  >
   412→                    <button
   413→                      onClick={() => setCartOpen(true)}
   414→                      className="max-w-7xl mx-auto w-full flex items-center justify-between bg-gradient-to-r from-orange-500 to-red-600 text-white px-6 py-4 rounded-2xl shadow-2xl shadow-orange-500/30 hover:shadow-orange-500/40 transition-shadow"
   415→                    >
   416→                      <div className="flex items-center gap-3">
   417→                        <div className="w-8 h-8 bg-white/20 rounded-lg flex items-center justify-center">
   418→                          <ShoppingCart className="w-4 h-4" />
   419→                        </div>
   420→                        <span className="font-semibold">
   421→                          {cartCount} article{cartCount > 1 ? "s" : ""}
   422→                        </span>
   423→                      </div>
   424→                      <div className="flex items-center gap-3">
   425→                        <span className="font-bold text-lg">{formatPrice(cartSubtotal)}</span>
   426→                        <ChevronRight className="w-5 h-5" />
   427→                      </div>
   428→                    </button>
   429→                  </motion.div>
   430→                )}
   431→              </AnimatePresence>
   432→            </motion.div>
   433→          )}
   434→
   435→          {step === "checkout" && (
   436→            <motion.div key="checkout" initial={{ opacity: 0, x: 40 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -40 }}>
   437→              <div className="max-w-2xl mx-auto px-4 sm:px-6 py-8">
   438→                <button
   439→                  onClick={() => setStep("menu")}
   440→                  className="flex items-center gap-2 text-gray-600 dark:text-gray-400 hover:text-orange-500 dark:hover:text-orange-400 mb-6 transition-colors"
   441→                >
   442→                  <ArrowLeft className="w-4 h-4" /> Retour au menu
   443→                </button>
   444→
   445→                <h2 className="text-2xl font-bold text-gray-900 dark:text-gray-100 mb-6 flex items-center gap-2">
   446→                  <ShoppingBag className="w-6 h-6 text-orange-500" /> Finaliser la commande
   447→                </h2>
   448→
   449→                <div className="space-y-6">
   450→                  {/* Guest info (if not logged in) */}
   451→                  {!customer && (
   452→                    <Card className="dark:bg-gray-800 dark:border-gray-700">
