     8→// ─── In-memory Rate Limiter for Seed Endpoint ───────────────────
     9→const seedRateLimitMap = new Map<string, { count: number; resetAt: number }>();
    10→const SEED_RATE_LIMIT_MAX = 3;       // max attempts per window
    11→const SEED_RATE_LIMIT_WINDOW = 60000; // 1 minute window
    12→
    13→function checkSeedRateLimit(ip: string): { allowed: boolean; retryAfterMs: number } {
    14→  const now = Date.now();
    15→  const entry = seedRateLimitMap.get(ip);
    16→
    17→  if (!entry || now > entry.resetAt) {
    18→    // No entry or window expired — start fresh
    19→    seedRateLimitMap.set(ip, { count: 1, resetAt: now + SEED_RATE_LIMIT_WINDOW });
    20→    return { allowed: true, retryAfterMs: 0 };
    21→  }
    22→
    23→  if (entry.count >= SEED_RATE_LIMIT_MAX) {
    24→    return { allowed: false, retryAfterMs: entry.resetAt - now };
    25→  }
    26→
    27→  entry.count += 1;
    28→  return { allowed: true, retryAfterMs: 0 };
    29→}

   586→export async function DELETE(request: Request) {
   587→  try {
   588→    const admin = await authenticateAdmin(request);
   589→    if (!admin) {
   590→      return NextResponse.json({ error: "Non autorisé" }, { status: 401 });
   591→    }
   592→    if (!hasRole(admin.role, ["admin"])) {
   593→      return NextResponse.json({ error: "Accès refusé" }, { status: 403 });
   594→    }
   595→
   596→    await db.$transaction([
   597→      db.expense.deleteMany(),
   598→      db.quote.deleteMany(),
   599→      db.invoice.deleteMany(),
   600→      db.review.deleteMany(),
   601→      db.order.deleteMany(),
   602→      db.reservation.deleteMany(),
   603→      db.staff.deleteMany(),
   604→      db.menuItem.deleteMany(),
   605→      db.driver.deleteMany(),
   606→      db.customer.deleteMany(),
   607→      db.admin.deleteMany(),
   608→      db.restaurantConfig.deleteMany(),
   609→      db.restaurant.deleteMany(),
   610→      db.platformAdmin.deleteMany(),
   611→    ]);
   612→
   613→    return NextResponse.json({ success: true, message: "Base de données vidée avec succès" });
   614→  } catch (error) {
